40 CLS:PRINT "TEST 1-6 ?";
50 G$=INKEY$:IF G$<>"" THEN 50
60 G$=INKEY$:IF G$="" THEN 60

80 IF G$="2" THEN 200
90 IF G$="3" THEN 300
92 IF G$="4" THEN 400
94 IF G$="5" THEN 500

520 IF G$="s" OR G$="S" THEN LET ZS=1
530 NEXT G:NEXT R:BEEP 3:GOTO 40

630 IF G<400 THEN 620
